name: Kesher Exact Marshmallow Video Task Resume

on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/kesher-exact-marshmallow-video-resume.yml'

permissions:
  actions: write
  contents: read

jobs:
  resume-exact-task:
    runs-on: ubuntu-latest
    steps:
      - name: Resume existing marshmallow Video Overview task
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          set -euo pipefail
          gh workflow run kesher-daily-video.yml --repo "$GITHUB_REPOSITORY" --ref main \
            -f operation=generate \
            -f target_slug=marshmallow-test-environmental-reliability
